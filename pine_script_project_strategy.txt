//@version=5
strategy(shorttitle="BB", title="Bollinger Bands", overlay=false,process_orders_on_close = true)//, timeframe="", timeframe_gaps=true)


length = input.int(20, minval=1)
maType = input.string("SMA", "Basis MA Type", options = ["SMA", "EMA", "SMMA (RMA)", "WMA", "VWMA"])
src = input(close, title="Source")
mult = input.float(2.0, minval=0.001, maxval=50, title="StdDev")

ma(source, length, _type) =>
    switch _type
        "SMA" => ta.sma(source, length)
        "EMA" => ta.ema(source, length)
        "SMMA (RMA)" => ta.rma(source, length)
        "WMA" => ta.wma(source, length)
        "VWMA" => ta.vwma(source, length)

basis = ma(src, length, maType)
dev = mult * ta.stdev(src, length)
upper = basis + dev
lower = basis - dev
offset = input.int(0, "Offset", minval = -500, maxval = 500)



// atrPeriod = input.int(10,    "ATR Length", minval = 1)
// factor =    input.float(3.0, "Factor",     minval = 0.01, step = 0.01)

// [supertrend, direction] = ta.supertrend(factor, atrPeriod)

// supertrend := barstate.isfirst ? na : supertrend


sell = low < lower //and low < supertrend and direction < 0 
buy = high  > upper //and high > supertrend and direction > 0 

float buy_sell_Signal_Line = 0

if(buy) 
    buy_sell_Signal_Line := 100000

if(sell)
    buy_sell_Signal_Line := -100000



if buy
    strategy.entry('Long', strategy.long,alert_message = 'buy at '+str.tostring(strategy.position_avg_price))

if sell
    strategy.entry('Short', strategy.short,alert_message = 'sell at '+str.tostring(strategy.position_avg_price))



plot(buy_sell_Signal_Line, 'BUY / SELL Signal Source', color=color.red, linewidth=2)
// Plosts
plotshape(strategy.position_size > 0 and strategy.position_size[1] < 0 , style=shape.arrowup, location=location.bottom, title='Buy', color=color.new(color.green, 0), size=size.tiny)
plotshape(strategy.position_size < 0 and strategy.position_size[1] > 0 , style=shape.arrowdown, location=location.top, title='Sell', color=color.new(color.red, 0), size=size.tiny)
